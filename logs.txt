    /**
    * Will invoke all of the callbacks that were registered to eventName with args

    function runEventCallbacks(eventName, args, isLocal){
      var callback, context, localOnly,
          eventCallbacks = (callbacks[eventName] || []).slice();
          eventCallbacks.sort(function (a, b){
            return a.rank - b.rank;
          });
/*ac:method:in*/
      for (var i = 0, l = eventCallbacks.length; i < l; i++){
        try{
          //This will apply the callback with all of the arguments@
          //given to emit except the event name.
          callback = eventCallbacks[i].cb;
          localOnly = eventCallbacks[i].localOnly;
          if(localOnly && !isLocal){
            continue;
          }
          context = eventCallbacks[i].ctx;
          if (typeof(args) != "object") {
            args = [];
          }
          callback.apply(context, args);
        }
        catch (e){
          reportError(e, callback);
        }
      }
    }
/*ac:method:end*/

    /**
    * Invokes the callbacks that were registered to Event name with args
    * @param eventName The name of the event to emit
    * @param args An array containing the arguments that should be passed
    *   to the callbacks.
    * @param sync whether to run the event in blocking mode (default: false)
    */
    function emitEventLocally(eventName, args, isLocal, sync) {
      if (sync === true) {
        runEventCallbacks(eventName, args, isLocal);
      }
      else {
        setTimeout(function(){runEventCallbacks(eventName, args, isLocal)},0);
      }
    }

    /**

    * @param eventName The name of the event to invoke.

    * arguments will be forwarded to the callbacks.

    function emitEvent(eventName, sync){
      if (eventName){
        var args = Array.prototype.slice.call(arguments,1);
        sync = (sync == true);

    /**
    * Bind a callback to eventName with the proper context (this)
    * @param eventName The name of the events callback will be binded to.
    * (Use the value in eventNames for default events)
    * @param callback A pointer to the function that should be invoked.
    * The function can take any number of arguemtns (depending on the event type)
    */
    function bindEvent(eventName, callback, localOnly, _rank) {
      if(eventAdapter[eventName] && typeof eventAdapter[eventName] === "function"){
        var Adapter = eventAdapter[eventName]({_eventName: eventName, _callback: callback, _localOnly: localOnly,  _rank: _rank});
        eventName = Adapter._eventName;
        callback = Adapter._callback;
        localOnly = Adapter._localOnly;
        _rank = Adapter._rank;
      }

      callbacks[eventName] = callbacks[eventName] || [];
      callbacks[eventName].push({
        cb: callback,
        ctx: this,
        localOnly: localOnly === true,
        rank : _rank ? _rank : _defaultEventRank
      });
    }

    function unbindEvent(eventName, callback) {
      for (var i = callbacks[eventName].length -1; i >= 0; i--) {
        if(callbacks[eventName][i].cb === callback) {
          callbacks[eventName].splice(i, 1);
        }
      }
    }

    function emptyEvent(eventName) {
      var eventCallbacks;

      if(callbacks[eventName] && callbacks[eventName].length > 0){
        eventCallbacks = callbacks[eventName].splice(0, callbacks[eventName].length);
      }
      return eventCallbacks || [];
    }

    function bindEventConditional(eventName, callback, localOnly, condition) {
      var localData = {};
      if (!mapToCondition[eventName]) {
        bindEvent(eventName, callback, localOnly);
      }
      else {
        var wrappedCallback = function() {
          mapToCondition[eventName](arguments, this, callback, condition, localData);
        };
        bindEvent(eventName, wrappedCallback, localOnly);
      }
    }

    function eventConditional_onCollectDone(args, context, callback, condition, data) {
      // Collectors engaged in page so far.
      if (!data.d) {
        data.d = [];
      }

      if (!condition) {
        condition = [];
      }

      var response = args[0];
      if (typeof(response) == "string") {
        response = JSON._parse(response);
      }

      var collectIsRelated = false;
      for (var index = 0; index < condition.length; index++) {
        if (response && typeof(response[condition[index]]) != "undefined") {
          if (Utils._indexInArray(data.d, condition[index]) < 0) {
            data.d.push(condition[index]);
          }
          collectIsRelated = true;
        }
      }

      if (collectIsRelated && data.d.length == condition.length) {
        // Call the original callback.
        callback.apply(context, args);
      }
    }

    /**
    * Checks whether the encoded collect data is big enough or not.
    * It'll later be treated differently based on this decision.
    * @param serializedData - encoded serializedData
    */
    function isRobustCollect(collectUrl, serializedData) {
      var collectUrlLength = collectUrl && collectUrl.length || 0,
          serializedDataLength = serializedData && serializedData.length || 0;
      if (encryptCollections && !UserAgent._isIE([6, 7])) {
        return (collectUrlLength + serializedDataLength) > maxGETLength && !flagCarbonCopyEnabled;
      }
      return serializedDataLength > 1000 && !flagCarbonCopyEnabled;
    }

    /**
    * @return The serlize form of the aggregated collect data -
    * RC4 encrypted if a collectEncryptionKey was defined
    * or base64 otherwise.
    * @param dataArray - The array that contains all the aggregated data

    function getSerializedAggregatedData(dataArray){
      var serializedData = encodeURIComponent(JSON._dump(dataArray));
      if (collectEncryptionKey){

      return encryptCollections ? CollectionEncryptor._encrypt(serializedData) : encodedSerializedData;
    }

    /**
    * Create and make a collect request to the server
    * @param serializedData - The serlized data to sent as created
    * by getSerializedAggregatedData function.
    */
    function makeCollectRequest(serializedData, cidsList, baseUrlOverride, collectCallbackReference, retryAttempt){
      var extraData = {};
      if (!retryAttempt){
        clearAggregatedCollectItem(cidsList);
      } else {
        clearAggregatedWithRetryCollectItem(cidsList, false);
      }

      // Send device cookies as query string parameters for CDN integrations
      if (sendDeviceCookiesInURL) {
        var deviceCookieCollected = SharedObject.syncGet(deviceCookieCollectedKey, false);
        if (!deviceCookieCollected) {
          addDeviceCookiesParams(extraData);
          SharedObject.set(deviceCookieCollectedKey, true, true);
        }
      }
      // Limit jsonp GET requests to short data. Most browsers support at least 2-4KB URI Length,
      // however, HTTP 1.1 specification suggests to avoid URI length of above 255 bytes, as some old browsers and proxies might not support it.
      var collectUrl = getCollectUrl(baseUrlOverride);
      if(isRobustCollect(collectUrl, serializedData)){
        var origin = win.location.protocol + "//" + win.location.host;
        var requestType = collectUrl.indexOf(origin) == 0 ? "ajax" : "xpost";
        sendRequest(requestType, collectUrl, {cid: cidsList}, collectCallbackReference, {d: serializedData}, extraData);
      }else{
        sendRequest(smallCollectRequestType, collectUrl, {d:serializedData, cid: cidsList}, collectCallbackReference, {}, extraData);
      }
    }

    /**
    * Collects the data. This function will:
    * 1) Add the data to the aggregated collect data structure
    * 2) Create a collection request if:
    *    a) The aggregated data has exceeded the limit.
    *    b) The immediate flag is set to true
    * If a collect request is not made the data will be saved in the aggregated
    * collect data structure and shared object (serlized)
    * awaiting the next collection.
    * @param collectorId - The id of the collector that is invoking this request.
    * @param collectedData - The collected data.
    * @param immediate - If set to true, a collect request will be made regardless
    * @param baseUrlOverride - override base URL
    * of the aggregated data size.
    */

    function collect(collectorId, collectedData, immediate, baseUrlOverride, skipCollect){
      /*ac:method:in*/
      if (collectorId) {
        if (!skipPreCollects[collectorId]) {
          emitEventLocally(eventNames._onPreCollect, [collectorId, collectedData], true, true);
        }
        if (collectedData) {
          if (aggregatedCollectWithRetryConf && aggregatedCollectWithRetryConf[collectorId]) {
            addAggregatedCollectDataWithRetry(aggregatedCollectDataWithRetry, collectorId, collectedData, aggregatedCollectWithRetryConf[collectorId] + 1);
            SharedObject.set(aggregatedDataWithRetryKey, aggregatedCollectDataWithRetry, true);
            SharedObject.set(aggregatedDataWithRetryCIDSKey, getCidsOfAggregatedData(aggregatedCollectDataWithRetry), true);
          }
          addAggregatedCollectItem(aggregatedCollectData, collectorId, collectedData);
        }
      }

      /*ac:method:in*/
      if (hasAggregatedData()){
        var serializedData = getSerializedAggregatedData(aggregatedCollectData);
        SharedObject.set(serializedDataKey, serializedData);
        var cidsList = getCidsOfAggregatedData(aggregatedCollectData);
        SharedObject.set(serializedDataCIDSKey, cidsList);
        if (!skipCollect && (immediate || serializedData.length >= aggregatedCollectDataCutoff)) {
          makeCollectRequest(serializedData, cidsList, baseUrlOverride, collectCallbackDelegate(collectorId));
        }
      }
      /*ac:method:in*/
    }
/*ac:method:end*/

    function sendRequest(type, url, params, callback, data, extra){
      type = type || smallCollectRequestType;
      params = params || {};
      var defaultParams = getDefaultParams();
/*ac:method:in*/
      if(typeof(params) == "string"){
        for (var key in defaultParams){
          if(Utils._getQueryParam(key, params) == undefined){
            params += "&" + key + "=" + defaultParams[key];
          }
        }
      }else{
        for (var key in defaultParams){
          params[key] = (typeof params[key] !== 'undefined') ? params[key] : defaultParams[key];
        }
      }
/*ac:method:in*/
      Net._request(type, url, params, callback, data, extra);
    }

        }

        attachEventRefs.push({obj: obj, type: type, func: handleEvent});
        return true;
      }
      return false;
    }
